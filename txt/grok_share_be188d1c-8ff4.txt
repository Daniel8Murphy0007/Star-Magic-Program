Implementation in JavaScript/HTML

<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>Pi to Solfeggio Frequencies</title>
</head>
<body>

    <input type="text" id="piInput" value="1415926535897932384626433832795" maxlength="100">
    <button onclick="playPiFrequencies()">Play</button>

    <script>
        // Define the 9 Solfeggio frequencies
        const solfeggioFrequencies = [174, 285, 396, 417, 528, 639, 741, 852, 963];

        // Function to map a digit (0-9) to a frequency
        function mapDigitToFrequency(digit) {
            digit = parseInt(digit);
            if (digit === 9) return solfeggioFrequencies[0]; // Map 9 to 174 Hz
            return solfeggioFrequencies[digit]; // Map 0-8 directly
        }

        // Function to play the sequence
        function playPiFrequencies() {
            const input = document.getElementById('piInput').value;
            if (!input || input.length > 100 || !/^\d+$/.test(input)) {
                alert("Please enter a valid string of up to 100 digits.");
                return;
            }

            const digits = input.split('').map(Number);
            const audioContext = new (window.AudioContext || window.webkitAudioContext)();
            const duration = 0.5; // Duration of each tone in seconds
            let currentTime = audioContext.currentTime;

            digits.forEach((digit, index) => {
                const frequency = mapDigitToFrequency(digit);
                const oscillator = audioContext.createOscillator();
                const gainNode = audioContext.createGain();

                oscillator.type = 'sine'; // Sine wave for pure tone
                oscillator.frequency.value = frequency;
                gainNode.gain.value = 0.3; // Volume control (0 to 1)

                oscillator.connect(gainNode);
                gainNode.connect(audioContext.destination);

                oscillator.start(currentTime + index * duration);
                oscillator.stop(currentTime + (index + 1) * duration);
            });
        }
    </script>
</body>
